   페이지 번호 만들기
   ===================================================== */

function createPagination() {

    pagination.innerHTML = "";

    const totalPages =
        Math.ceil(
            titles.length / itemsPerPage
        );


    for (
        let page = 1;
        page <= totalPages;
        page++
    ) {

        const button =
            document.createElement("button");

        button.className =
            "page-number";

        button.textContent =
            page;


        button.onclick = function() {

            showPage(page);

        };


        pagination.appendChild(button);

    }

}


/* =====================================================
   페이지 표시
   ===================================================== */

function showPage(page) {

    const start =
        (page - 1) * itemsPerPage;

    const end =
        start + itemsPerPage;


    /* 제목 표시 */

    titles.forEach(
        (title, index) => {

            if (
                index >= start &&
                index < end
            ) {

                title.style.display =
                    "block";

            } else {

                title.style.display =
                    "none";

            }

        }
    );


    /* 페이지 번호 활성화 */

    const buttons =
        document.querySelectorAll(
            ".page-number"
        );


    buttons.forEach(
        (button, index) => {

            if (
                index + 1 === page
            ) {

                button.classList.add(
                    "active"
                );

            } else {

                button.classList.remove(
                    "active"
                );

            }

        }
    );


    /* 해당 페이지 첫 번째 시 */

    showContent(start);

}


/* =====================================================
   ★ 제목 클릭
   본문 표시 + 음악 재생
   ===================================================== */

function showContent(number) {


    /* 모든 본문 닫기 */

    contents.forEach(
        content => {

            content.classList.remove(
                "active"
            );

        }
    );


    /* 선택한 본문 열기 */

    if (contents[number]) {

        contents[number].classList.add(
            "active"
        );


        /*
           새로운 시를 열면
           본문 맨 위에서 시작
        */

        contents[number].scrollTop = 0;

    }


    /* 제목 활성화 */

    titles.forEach(
        (title, index) => {

            if (index === number) {

                title.classList.add(
                    "active"
                );

            } else {

                title.classList.remove(
                    "active"
                );

            }

        }
    );


    /* =================================================
       제목을 클릭하면 음악 재생
       ================================================= */

    if (music) {

        music.play().catch(
            function(error) {

                console.log(
                    "음악 재생을 시작하지 못했습니다.",
                    error
                );

            }
        );

    }

}
